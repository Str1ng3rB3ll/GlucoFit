Methode 1: Dexcom Live-Feed URL Vollautomatisch

Verbinde deine kostenlose Dexcom Share Proxy-URL (Cloudflare Worker) oder Nightscout-URL (/api/v1/entries.json).

Bereit zum Abrufen.

So richtest du den Dexcom G7 Server in 2 Min. gratis ein:

1. In der Dexcom G7 App unter Verbindungen → Share einen Follower einladen (deine E-Mail).

2. Auf dash.cloudflare.com kostenlos anmelden → Workers & Pages → Create Worker.

3. Diesen Code einfügen, oben deinen Dexcom-Benutzernamen & Passwort eintragen und auf Deploy klicken.

4. Die *.workers.dev URL oben als Live-Feed URL speichern!

// Cloudflare Worker Proxy für Dexcom Share EU (shareous1.dexcom.com)
const DEXCOM_USER = "DEIN_DEXCOM_BENUTZERNAME";
const DEXCOM_PASS = "DEIN_DEXCOM_PASSWORT";
const DEXCOM_APP_ID = "d8665ade-9673-4e27-9ff6-92db4ce13d13"; // Dexcom EU App ID

export default {
  async fetch(request) {
    const cors = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type",
      "Content-Type": "application/json"
    };
    if (request.method === "OPTIONS") return new Response(null, { headers: cors });

    try {
      const base = "https://shareous1.dexcom.com/ShareWebServices/Services";
      // 1. Account ID abrufen
      const authRes = await fetch(`${base}/General/AuthenticatePublisherAccount`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          accountName: DEXCOM_USER,
          password: DEXCOM_PASS,
          applicationId: DEXCOM_APP_ID
        })
      });
      const accountId = await authRes.json();

      // 2. Session ID abrufen
      const loginRes = await fetch(`${base}/General/LoginPublisherAccountById`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          accountId,
          password: DEXCOM_PASS,
          applicationId: DEXCOM_APP_ID
        })
      });
      const sessionId = await loginRes.json();

      // 3. Aktuellsten Dexcom G7 Wert abrufen
      const readRes = await fetch(
        `${base}/Publisher/ReadPublisherLatestGlucoseValues?sessionId=${sessionId}&minutes=1440&maxCount=1`,
        { method: "POST", headers: { "Content-Type": "application/json" } }
      );
      const entries = await readRes.json();
      const latest = entries?.[0];

      if (!latest) {
        return new Response(JSON.stringify({ error: "Keine aktuellen Werte" }), { status: 404, headers: cors });
      }

      const timestamp = parseInt(String(latest.WT).replace(/\D/g, ""), 10) || Date.now();
      return new Response(
        JSON.stringify({
          sgv: latest.Value,
          trend: latest.Trend,
          date: timestamp
        }),
        { headers: cors }
      );
    } catch (err) {
      return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: cors });
    }
  }
};
